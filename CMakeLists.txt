cmake_minimum_required(VERSION 3.16)
project(sdr_touch LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt5 REQUIRED COMPONENTS Widgets Multimedia)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SOAPY REQUIRED IMPORTED_TARGET SoapySDR)
pkg_check_modules(FFTW REQUIRED IMPORTED_TARGET fftw3f)
pkg_check_modules(ICU REQUIRED IMPORTED_TARGET icu-i18n)
find_library(WDSP_LIBRARY wdsp REQUIRED)
find_path(WDSP_INCLUDE_DIR wdsp.h REQUIRED)

add_library(signal_processing dsp.cpp)
target_include_directories(signal_processing PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} ${WDSP_INCLUDE_DIR})
target_link_libraries(signal_processing PUBLIC PkgConfig::FFTW ${WDSP_LIBRARY})
target_compile_options(signal_processing PRIVATE -Wall -Wextra -Wpedantic)

add_library(scan_log scan_log.cpp station_lists.cpp)
target_link_libraries(scan_log PUBLIC Qt5::Core)
target_compile_options(scan_log PRIVATE -Wall -Wextra -Wpedantic)

add_library(digital_support geography.cpp digital.cpp digital.hpp adsb.cpp adsb.hpp voice.cpp voice.hpp aprs.cpp aprs.hpp morse.cpp morse.hpp)
target_link_libraries(digital_support PUBLIC Qt5::Core PkgConfig::ICU)
target_compile_options(digital_support PRIVATE -Wall -Wextra -Wpedantic)

add_executable(sdr-touch main.cpp receiver.cpp touch_keyboard.cpp arrows.qrc)
target_link_libraries(sdr-touch PRIVATE signal_processing scan_log digital_support Qt5::Widgets Qt5::Multimedia PkgConfig::SOAPY)
target_compile_options(sdr-touch PRIVATE -Wall -Wextra -Wpedantic)

enable_testing()
add_executable(signal-tests tests.cpp)
target_link_libraries(signal-tests PRIVATE signal_processing)
add_test(NAME signal-tests COMMAND signal-tests)
add_executable(scan-tests scan_tests.cpp)
target_link_libraries(scan-tests PRIVATE scan_log)
add_test(NAME scan-tests COMMAND scan-tests)
add_executable(station-lists-tests station_lists_tests.cpp)
target_link_libraries(station-lists-tests PRIVATE scan_log)
add_test(NAME station-lists-tests COMMAND station-lists-tests)
add_executable(digital-tests digital_tests.cpp)
target_link_libraries(digital-tests PRIVATE digital_support)
add_test(NAME digital-tests COMMAND digital-tests)
add_executable(adsb-tests adsb_tests.cpp)
target_link_libraries(adsb-tests PRIVATE digital_support)
add_test(NAME adsb-tests COMMAND adsb-tests)
add_executable(digital-replay-test digital_replay_test.cpp)
target_link_libraries(digital-replay-test PRIVATE digital_support signal_processing)
add_executable(voice-replay-test voice_replay_test.cpp)
target_link_libraries(voice-replay-test PRIVATE digital_support)
add_executable(voice-tests voice_tests.cpp)
target_link_libraries(voice-tests PRIVATE digital_support)
add_test(NAME voice-tests COMMAND voice-tests)
find_program(MULTIMON_NG_EXECUTABLE multimon-ng)
if(MULTIMON_NG_EXECUTABLE)
    add_executable(aprs-tests aprs_tests.cpp)
    target_link_libraries(aprs-tests PRIVATE digital_support signal_processing)
    add_test(NAME aprs-tests COMMAND aprs-tests)
    add_executable(morse-tests morse_tests.cpp)
    target_link_libraries(morse-tests PRIVATE digital_support signal_processing)
